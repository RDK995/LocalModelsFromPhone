```ts
export interface PassageSettings {
  passageMinWords: number;     // default 150
  passageMaxWords: number;     // default 400
  noteExcerptTokens: number;   // default 2500  (excerpt size cap, tokens estimated as ceil(chars/4))
  noteMinWords: number;        // default 40    (fewer words of page text -> "empty")
  noteMinRelevance: number;    // default 0     (best passage BM25 score must be > this, else "empty")
}
export const DEFAULT_PASSAGE_SETTINGS: PassageSettings;

export function estimateTokens(s: string): number;          // Math.ceil(s.length / 4)
export function tokenize(s: string): string[];               // lowercase, split on non-letter/digit
                                                             // (Unicode-aware: /[^\p{L}\p{N}]+/u),
                                                             // drop empty tokens and a small English
                                                             // stopword list (the, a, an, and, or, of,
                                                             // to, in, on, for, is, are, was, were, be,
                                                             // with, by, at, as, it, this, that, what,
                                                             // how, why, which, who, do, does, from)
export function splitPassages(text: string, minWords: number, maxWords: number): string[];
export function rankPassages(passages: string[], terms: string[]): Array<{ index: number; score: number }>;
export const BOT_CHALLENGE_MARKERS: readonly string[];
export function isBotChallenge(text: string): boolean;

export type NoteExcerpt =
  | { skip: "empty" | "blocked"; reason: string }
  | { skip: null; excerpt: string; passages: number; tokens: number };

export function buildNoteExcerpt(
  page: { title: string; text: string },
  terms: string,                 // the sub-question and the main question joined by a space
  settings?: Partial<PassageSettings>
): NoteExcerpt;
```

Behaviour:
- splitPassages: split the text into blocks on blank lines (one or more empty/whitespace-only lines)
  and on heading lines (a line starting with 1-6 `#` then a space, or a short line of <= 12 words
  with no terminal punctuation that is followed by a blank line or longer text — keep the heading
  rule simple; markdown `#` headings are required, the plain-text heuristic is optional). A heading
  starts a new passage and is kept at the start of it. Consecutive blocks are merged into one passage
  until it reaches at least `minWords` words; a passage never crosses a heading. A block longer than
  `maxWords` is cut into chunks of at most `maxWords` words at word boundaries (prefer a sentence end
  within the last 25% of the chunk). Trailing passages shorter than minWords are allowed (a short
  page is one short passage). Whitespace inside a passage is preserved as single spaces/newlines; the
  passage text must be a verbatim substring of the page text after whitespace normalisation
  (`s.replace(/\s+/g, " ").trim()`), so quotes taken from it still match the page.
- rankPassages: Okapi BM25 over the page's own passages as the corpus, k1 = 1.2, b = 0.75,
  idf = ln(1 + (N - n + 0.5) / (n + 0.5)) (always positive). Query terms = unique tokenize(terms).
  Returns every passage index with its score, sorted by score descending, ties by lower index first.
- isBotChallenge: true when the lowercased, whitespace-normalised text contains any marker AND the
  page has fewer than 400 words (a long article that merely mentions CAPTCHAs is not blocked).
  Markers (lowercase): "captcha", "verify you are human", "are you a robot", "are you human",
  "checking your browser", "checking if the site connection is secure",
  "enable javascript and cookies to continue", "attention required! | cloudflare",
  "unusual traffic from your computer", "access denied", "cf-ray", "ddos protection by".
- buildNoteExcerpt, in this order:
  1. isBotChallenge(title + "\n" + text) -> `{ skip: "blocked", reason }`.
  2. Word count of text (tokens split on whitespace) < noteMinWords -> `{ skip: "empty", reason }`.
  3. Rank passages; if there are no passages or the best score <= noteMinRelevance ->
     `{ skip: "empty", reason }` (reason says no passage matched the research terms).
  4. Otherwise build the excerpt:
     ```
     Title: <title>

     First paragraph:
     <first paragraph>

     Relevant passages:
     <passage>

     <passage>
     ```
     First paragraph = the first blank-line-separated block of the text that is not a heading line
     (whitespace-normalised). Take passages in rank order, only those with score > noteMinRelevance,
     skipping any passage whose normalised text equals the first paragraph; add each while the whole
     excerpt's estimateTokens stays <= noteExcerptTokens; emit the chosen passages in page order.
     If not even the best passage fits whole, include it cut at a word boundary to fit. If the first
     paragraph alone would exceed half the cap, cut it at a word boundary to half the cap. The
     returned excerpt's estimateTokens must ALWAYS be <= noteExcerptTokens and it always contains
     at least one passage (whole or cut). `passages` = number of passages included, `tokens` =
     estimateTokens(excerpt).
